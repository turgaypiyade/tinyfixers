 public static Queue<Action> pending=new Queue<Action>();
 public static Task Write(List<KeyValuePair<string,Dictionary<string,object>>> values){writes++;var t=new TaskCompletionSource<bool>();Action commit=()=>{if(failWrites){t.SetException(new Exception("offline"));return;}foreach(var kv in values){var id=kv.Key.Split('/').Last();DocumentSnapshot d;if(!docs.TryGetValue(kv.Key,out d))docs[kv.Key]=d=new DocumentSnapshot{Id=id};foreach(var field in kv.Value)d.Data[field.Key]=field.Value;}t.SetResult(true);};if(holdWrites)pending.Enqueue(commit);else commit();return t.Task;}
 public static void Release(){while(pending.Count>0)pending.Dequeue()();}
}
