public static class Checks{
 static int checks;static void Check(bool b,string why){checks++;if(!b)throw new Exception(why);}
 static void BaseUnchanged(LevelData l,int[] ids,int[] origins){Check(l.obstacles.SequenceEqual(ids)&&l.obstacleOrigins.SequenceEqual(origins),"Overlay edits must preserve every authored underlying lantern");}
 public static void Run(bool before){
  foreach(var id in new[]{ObstacleId.Wall,ObstacleId.MetalWall}){
   foreach(var order in new[]{new[]{13,14,15},new[]{15,14,13},new[]{14,15,13},new[]{14,13,15}}){
    var l=new LevelData();var e=new Editor();var ids=(int[])l.obstacles.Clone();var origins=(int[])l.obstacleOrigins.Clone();
    foreach(int i in order)e.Paint(l,i,id);
    if(before){Check(l.stackedObstacles.All(s=>s.wallPieceId==0&&s.wallPieceOriginPlusOne==0)&&Editor.Count(l,id)==3,"Old editor must leave manually painted wall cells ungrouped");Console.WriteLine("REPRODUCED: existing Overlay wall brush paints three independent caps despite using one piece number.");return;}
    Check(l.stackedObstacles.Length==3&&l.stackedObstacles.All(s=>s.wallPieceId==1&&s.wallPieceOriginPlusOne==14),"Manual brush must join adjacent cells, regardless of paint order");
    Check(Editor.Count(l,id)==1,"Manual wall route must count one goal");BaseUnchanged(l,ids,origins);
    e.Paint(l,14,id);Check(l.stackedObstacles.Length==3,"Repainting cannot duplicate a layer");
    e.Erase(l,14,id);Check(Editor.Count(l,id)==2,"Removing a bridge must split the route into two live pieces");
    Check(l.stackedObstacles.All(s=>s.wallPieceOriginPlusOne==s.originCellIndex+1),"Split pieces must have actual surviving origins");BaseUnchanged(l,ids,origins);
    e.Paint(l,14,id);Check(Editor.Count(l,id)==1,"Restoring a bridge must reconnect both pieces");
    e.Erase(l,13,id);Check(l.stackedObstacles.All(s=>s.wallPieceOriginPlusOne==15),"Deleting an origin must re-anchor surviving cells");
    e.Paint(l,14,id,2);Check(Editor.Count(l,id)==2,"Changing piece number separates adjacent cells");
    e.Paint(l,14,id,1);Check(Editor.Count(l,id)==1,"Repainting original number reconnects cells");
    Check(Editor.Next(l)==2,"New Piece must inspect overlay numbers");
    e.Paint(l,30,id,9);Check(Editor.Next(l)==10&&Editor.Count(l,id)==2,"Disconnected brush islands are independent; next number includes overlay-only pieces");
    BaseUnchanged(l,ids,origins);
   }
  }
  {
   var l=new LevelData();var e=new Editor();e.Paint(l,13,ObstacleId.Wall);e.Paint(l,14,ObstacleId.MetalWall);
   Check(l.stackedObstacles.All(s=>s.wallPieceOriginPlusOne==s.originCellIndex+1),"Wall and MetalWall must never join");
   var d=new LevelData();e.Paint(d,5,ObstacleId.Wall);e.Paint(d,6,ObstacleId.Wall);Check(Editor.Count(d,ObstacleId.Wall)==2,"Row edges must not wrap into neighbors");
   d.cells[8]=0;int count=d.stackedObstacles.Length;e.Paint(d,8,ObstacleId.Wall);Check(d.stackedObstacles.Length==count,"Wall brush cannot stamp a hole");
  }
  {
   var l=new LevelData();var e=new Editor();foreach(int i in new[]{13,14,15}){
    l.wallPieceIds[i]=1;l.stackedObstacles=l.stackedObstacles.Concat(new[]{new StackedObstacleEntry{originCellIndex=i,obstacleId=ObstacleId.Wall,wallPieceOriginPlusOne=14}}).ToArray();
   }
   Editor.BaseRecompute(l);Check(l.wallPieceIds.All(i=>i==0),"Base-only origin rebuilding may clear its own array");
   Check(l.stackedObstacles.All(s=>s.wallPieceId==1&&s.wallPieceOriginPlusOne==14),"Already-authored226-style overlay numbers must survive base edits");
   e.Erase(l,13,ObstacleId.Wall);Check(l.stackedObstacles.All(s=>s.wallPieceOriginPlusOne==15),"Legacy authored grouped route must remain editable after origin removal");
   var legacy=new LevelData();legacy.stackedObstacles=new[]{new StackedObstacleEntry{originCellIndex=13,obstacleId=ObstacleId.Wall},new StackedObstacleEntry{originCellIndex=14,obstacleId=ObstacleId.Wall}};
   Editor.Normalize(legacy);Check(legacy.stackedObstacles.All(s=>s.wallPieceId==0&&s.wallPieceOriginPlusOne==0),"Unnumbered legacy wall caps must remain independent");
  }
  {
   var l=new LevelData();var e=new Editor();e.Paint(l,13,ObstacleId.Wall);e.Paint(l,14,ObstacleId.Wall);
   e.Paint(l,13,ObstacleId.Grass);e.Erase(l,13,ObstacleId.Wall);
   Check(l.stackedObstacles.Any(s=>s.originCellIndex==13&&s.obstacleId==ObstacleId.Grass),"Wall-only eraser must retain other overlays");
   Check(l.obstacles[13]==(int)ObstacleId.Lantern&&l.obstacleOrigins[13]==13,"Wall-only eraser must retain base lantern");
   e.EraseAll(l,14);Check(l.stackedObstacles.All(s=>s.originCellIndex!=14)&&l.obstacles[14]==(int)ObstacleId.Lantern,"Existing all-overlay eraser still removes overlay entries only");
  }
  {
   var l=new LevelData();var e=new Editor();e.Paint(l,13,ObstacleId.Wall);e.Paint(l,14,ObstacleId.Wall);
   Check(!Editor.Cover(l,13,ObstacleId.Stone),"Opaque one-cell layer cannot partially cover a grouped wall");
   Check(Editor.Cover(l,13,ObstacleId.Grass),"Existing transparent grass cover remains allowed");
   e.Draw(l);Check(GUI.labels.Count==2&&GUI.labels.All(s=>s=="1")&&e.previews.Count==4,"Wall preview must show numbered wall cells and underlying lantern windows");
   e.Paint(l,13,ObstacleId.Grass);e.Draw(l);Check(GUI.labels.Count==2,"Grass must not hide the editor's wall footprint");
   var hidden=new LevelData();e.Paint(hidden,13,ObstacleId.Wall);hidden.stackedObstacles=hidden.stackedObstacles.Concat(new[]{new StackedObstacleEntry{originCellIndex=13,obstacleId=ObstacleId.Stone,stackOrder=10}}).ToArray();
   e.Draw(hidden);Check(GUI.labels.Count==0,"A higher opaque overlay must not be painted over by the wall preview");
  }
  {
   var l=new LevelData();var e=new Editor();e.Paint(l,13,ObstacleId.Stone);e.Paint(l,13,ObstacleId.Stone);
   Check(l.stackedObstacles.Length==1&&l.stackedObstacles[0].wallPieceId==0&&l.stackedObstacles[0].wallPieceOriginPlusOne==0,"Existing working stone overlay and deduplication remain unchanged");
   e.Paint(l,13,ObstacleId.Grass);Check(l.stackedObstacles.Length==2&&l.stackedObstacles[1].stackOrder>l.stackedObstacles[0].stackOrder,"Existing overlay stacking order remains unchanged");
   var c=new LevelData();foreach(int i in new[]{7,8,13,14}){c.obstacles[i]=(int)ObstacleId.ColorChest;c.obstacleOrigins[i]=7;}
   e.Paint(c,7,ObstacleId.Wall);Check(c.stackedObstacles.Length==0,"Wall brush must retain the existing full-cover restriction over a2x2 chest");
  }
  Console.WriteLine("PASS: "+checks+" production editor paint/group/erase/cover/draw checks; actual library footprints, no Unity/board simulation.");
 }
