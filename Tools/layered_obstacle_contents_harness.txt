public static class Checks {
 static int checks;static void Check(bool ok,string reason){checks++;if(!ok)throw new Exception(reason);}
 public static void Run(bool before){
  var lib=new Library();
  foreach(var id in new[]{ObstacleId.GelLauncherUp,ObstacleId.GelLauncherDown,ObstacleId.GelLauncherLeft,ObstacleId.GelLauncherRight,ObstacleId.LaserCannon,ObstacleId.RhodesStatue,ObstacleId.TarpStatue,ObstacleId.PaintCanBox}){
   var def=lib.Get(id);def.size=GelLauncherFx.IsVertical(id)?new Vector2Int(1,2):new Vector2Int(2,1);
   foreach(int saved in before?new[]{def.hits}:Enumerable.Range(1,def.hits)){
    var s=new State();s.Cover(id,saved);var p=new Spawner{board=new BoardController{ObstacleStateService=s},resolvedLevel=s.level};var image=new Image();
    s.level.obstacles[4]=(int)id; // Actual beneath drawing stamps identity, NOT the cover HP.
    p.Configure(image,def);
    if(BoardController.IsGelLauncher(id)){
     var view=image.GetComponent<GelLauncherPressView>();
     if(before){Check(Math.Abs(view.LiquidRatio-.385f)<.00001,"legacy cover HP must reproduce a half-empty press");Console.WriteLine("REPRODUCED: covered fresh launcher reads Grass HP1; actual Configure/ApplyPose draws38.5% liquid instead of63%.");return;}
     float expected=saved==1?.385f:.63f;Check(Math.Abs(view.LiquidRatio-expected)<.00001,"press liquid must match its own saved layer");Check(!image.enabled,"assembled press stays on child visuals, not the generic root");
     s.Expose(id,saved);p.Configure(image,def);Check(image.GetComponent<GelLauncherPressView>()==view&&Math.Abs(view.LiquidRatio-expected)<.00001,"promotion reuses children and cannot refill actual damage");
    }else{
     int hits=id==ObstacleId.LaserCannon?image.GetComponent<LaserCannonView>().remaining:id==ObstacleId.RhodesStatue?image.GetComponent<RhodesStatueView>().remaining:id==ObstacleId.TarpStatue?image.GetComponent<TarpStatueView>().hits:image.GetComponent<PaintCanBoxView>().remaining;
     Check(hits==saved,"secondary visual must use its saved HP rather than the cover HP");
    }
    Check(p.Remaining(def)==saved,"shared resolver preserves the layer count");
    Check(s.remainingHitsByOrigin[4]==(BoardController.IsGelLauncher(id)?saved:1),"presentation cannot mutate gameplay HP");
   }
  }
  var fallback=new Spawner();Check(fallback.Remaining(lib.Get(ObstacleId.LaserCannon))==4&&fallback.Remaining(lib.Get(ObstacleId.LaserCannon),2)==2,"no-state fallback respects authored damaged count");
  var zero=new State();zero.Expose(ObstacleId.PaintCanBox,0);Check(new Spawner{board=new BoardController{ObstacleStateService=zero},resolvedLevel=zero.level}.Remaining(lib.Get(ObstacleId.PaintCanBox))==0,"known exhausted layer does not refill");
  var ward=new WardrobeObstacleView();ward.SetClosedSprite(new Sprite());var wp=new Spawner{board=new BoardController()};wp.OpenWardrobe(ward,lib.Get(ObstacleId.Wardrobe));Check(ward.items==8&&ward.doors==1,"actual wardrobe open route installs all eight sprite items");
  Console.WriteLine("PASS: "+checks+" production layered-content/press-pose/wardrobe assertions; four launcher directions and real partial states preserved.");
 }
