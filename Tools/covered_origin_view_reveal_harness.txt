public static class Tests {
 static int checks,orders;static void Check(bool ok,string reason){checks++;if(!ok)throw new Exception(reason);}
 static IEnumerable<int[]> Permute(int[] cells){if(cells.Length==0){yield return cells;yield break;}foreach(int cell in cells)foreach(var tail in Permute(cells.Where(c=>c!=cell).ToArray()))yield return new[]{cell}.Concat(tail).ToArray();}
 public static void Run(bool before){
  // Same origin/footprint as level 212's left PaintCanBox; exercise all callback orders.
  foreach(var order in before?new[]{new[]{83,74,75,84}}:Permute(new[]{74,75,83,84})){
   var level=new Level();var state=new State{level=level};var spawn=new Spawner{resolvedLevel=level,board=new BoardController{ObstacleStateService=state}};state.Connect(spawn);
   var grass=new Dictionary<int,Image>();foreach(int cell in order){level.obstacles[cell]=(int)ObstacleId.Grass;level.obstacleOrigins[cell]=cell;grass[cell]=spawn.Add(cell,ObstacleId.Grass);}
   var paint=new Image();spawn.beneathViewsByCell[74]=paint;
   foreach(int cell in order){
    // ClearObstacleFromLevel restores data, retires the cover, then requests the revealed origin.
    level.obstacles[cell]=(int)ObstacleId.PaintCanBox;level.obstacleOrigins[cell]=74;
    spawn.Retire(cell,ObstacleId.Grass);state.Reveal(74,ObstacleId.PaintCanBox);
    if(before&&cell==74){
     Check(!grass[74].gameObject.destroyed,"old gate must reproduce an orphaned grass image");
     Check(paint.gameObject.destroyed,"old gate must reproduce paint being retired as grass");
     Console.WriteLine("REPRODUCED: non-origin reveal replaces the tracked grass with paint; origin hit deletes paint and leaves grass orphaned.");return;
    }
    if(before)continue;
    Check(grass[cell].gameObject.destroyed,"retired grass must not survive as a ghost");
    if(level.obstacles[74]==(int)ObstacleId.Grass){
     Check(state.events==0&&state.initializations==0,"still-covered origin must not emit/init");
     Check(spawn.beneathViewsByCell[74]==paint&&!paint.gameObject.destroyed,"covered paint preview must remain underneath");
     Check(spawn.obstacleViewsByOrigin[74]==grass[74],"live grass ownership must stay intact");
     Check(spawn.groundCalls==0,"suppressed reveal must not do renderer/ground work");
    }else{
     Check(spawn.obstacleViewsByOrigin[74]==paint&&spawn.obstacleDefsByOrigin[74].id==ObstacleId.PaintCanBox,"revealed paint must own the existing image");
     Check(!paint.gameObject.destroyed&&!spawn.beneathViewsByCell.ContainsKey(74),"promoted preview must remain alive");
    }
   }
   Check(spawn.newViews==0,"reveal must reuse the preview without extra images");
   Check(spawn.configurations==1,"only the actual preview promotion may refresh its contents");
   spawn.Retire(74,ObstacleId.PaintCanBox);spawn.Retire(74,ObstacleId.PaintCanBox);
   Check(paint.gameObject.destroyed&&spawn.obstacleViewsByOrigin.Count==0&&spawn.obstacleDefsByOrigin.Count==0&&spawn.beneathViewsByCell.Count==0,"final retirement must leave no tracked/preview images");orders++;
  }
  var s=new State();s.Reveal(0,ObstacleId.PaintCanBox);Check(s.initializations==0,"null level is ignored");
  s.level=new Level();s.Reveal(-1,ObstacleId.PaintCanBox);s.Reveal(99,ObstacleId.PaintCanBox);Check(s.initializations==0,"invalid origins are ignored");
  s.level.obstacles[74]=(int)ObstacleId.PaintCanBox;s.level.obstacleOrigins[74]=75;s.Reveal(74,ObstacleId.PaintCanBox);Check(s.initializations==0,"same ID with a different origin is ignored");
  s.level.obstacleOrigins[74]=74;s.Reveal(74,ObstacleId.PaintCanBox,5);Check(s.initializations==1&&s.lastRemaining==5,"real reveal preserves the remaining-hit override");
  s.level.obstacles[74]=(int)ObstacleId.Safe;s.Reveal(74,ObstacleId.Safe,3);Check(s.initializations==2&&s.lastRemaining==3,"gate also allows exposed non-paint layers");
  Console.WriteLine("PASS: "+checks+" production reveal/view ownership checks across "+orders+" cover-clear orders (including both vertical directions).");
 }
